  tag: ''
  pullPolicy: IfNotPresent

  create: true
  name: ''
  annotations: {}

    # eks.amazonaws.com/role-arn: arn:aws:iam::ACCOUNT:role/acme-health-backend

  type: ClusterIP
  port: 80
  targetPort: 3001

ingress:
  enabled: true
  className: nginx
  host: acme-health-backend.example.com
  tls:
    enabled: true
    secretName: acme-health-backend-tls
  annotations:
    cert-manager.io/cluster-issuer: letsencrypt-prod
    nginx.ingress.kubernetes.io/proxy-body-size: 8m

  requests:
    cpu: 100m
    memory: 128Mi

    # quota rather than letting it use idle capacity. Memory IS limited, because exceeding
